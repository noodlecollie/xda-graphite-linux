# Make sure we log to see what's going on.
log haret.txt

# The kernel file we are booting should be named "zImage",
# and should be located next to HaRET.exe.
set KERNEL zImage

# This is "Generic Intel PXA27x" (pxa27x)
# https://www.arm.linux.org.uk/developer/machines/list.php?mid=660
# It expired in 2005.
set MTYPE 660

# Enables logging on device's framebuffer after HaRET takes over
# CPU control, and stops logging to file.
set FBDURINGBOOT 1

set cmdline "console=tty0 panic=20 root=/dev/mmcblk0p2 init=/usr/bin/init"
boot
